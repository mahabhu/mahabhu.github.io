\def\addressSep{$\diamond$} % Separator between fields on one address line

\def\name#1{\def\@name{#1}}

% Up to three address lines, printed in the order they are declared
\let\@addressone\relax
\let\@addresstwo\relax
\let\@addressthree\relax

\def\address#1{%
  \@ifundefined{@addressone}{\def\@addressone{#1}}{%
  \@ifundefined{@addresstwo}{\def\@addresstwo{#1}}{%
                             \def\@addressthree{#1}}}%
}

% Style one address line; \\ inside it becomes the field separator
\def\printaddress#1{%
  \begingroup
    \def \\ {\addressSep\ }
    \centerline{#1}%
  \endgroup
  \par
  \addressskip
}

\def\printname{%
  \begingroup
    \hfil{\MakeUppercase{\namesize\bf \@name}}\hfil
    \nameskip\break
  \endgroup
}

% Prints the name followed by every address line that was declared
\newcommand{\makeheader}{%
  \printname
  \@ifundefined{@addressone}{}{\printaddress{\@addressone}}%
  \@ifundefined{@addresstwo}{}{\printaddress{\@addresstwo}}%
  \@ifundefined{@addressthree}{}{\printaddress{\@addressthree}}%
}

\makeatother

%----------------------------------------------------------------------------------------
%	SECTION FORMATTING
%----------------------------------------------------------------------------------------

% One argument - the section name.
% The \nobreak/\@beginparpenalty chain below keeps the title, the rule and the
% first entry on the same page - without it a section heading can be left
% stranded at the foot of a page while its content starts on the next one.
\makeatletter
\newenvironment{rSection}[1]{%
  \sectionskip
  \MakeUppercase{\bf #1} % Section title
  \par\nobreak            % end the title line; forbid a break before the rule
  \sectionlineskip
  \nobreak
  \hrule % Horizontal line
  \nobreak
  \@beginparpenalty\@M    % forbid a break between the rule and the first entry
  \begin{list}{}{%
    \setlength{\leftmargin}{1.5em} % Margin within the section
  }
  \item[]
}{%
  \end{list}
}
\makeatother

%----------------------------------------------------------------------------------------
%	POSITION / EXPERIENCE FORMATTING
%----------------------------------------------------------------------------------------

% Four arguments - organization, location, title, dates.
% The whole subsection is a single paragraph and \item is just a line break, so
% every line sits exactly one \baselineskip below the previous one. A nested list
% would insert \topsep/\parsep/\itemsep glue and break that rhythm.
\newenvironment{rSubsection}[4]{%
 \def\item{\\}% Each \item line is a plain new line, no bullet, no extra leading
 {\bf #1} \hfill {#2}% Bold organization, location on the right
 \ifthenelse{\equal{#3}{}}{}{ % Skip the title line if the third argument is empty
  \\
  {\em #3} \hfill \dates{#4}% Italic title, italic dates on the right
  }%
}{%
}
